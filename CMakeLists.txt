cmake_minimum_required(VERSION 3.24)
project(tree_lab VERSION 1.0 LANGUAGES CXX)

add_library(tree_lab INTERFACE)
target_include_directories(tree_lab INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(tree_lab INTERFACE cxx_std_20)

if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    set(tree_lab_standalone ON)
else()
    set(tree_lab_standalone OFF)
endif()
option(TREE_LAB_BUILD_EXAMPLES "Build interactive demo" ${tree_lab_standalone})
option(TREE_LAB_BUILD_TESTS "Build GoogleTest tests" ${tree_lab_standalone})
option(TREE_LAB_BUILD_BENCHMARKS "Build benchmark" ${tree_lab_standalone})
option(TREE_LAB_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

function(tree_lab_configure_target target)
    target_link_libraries(${target} PRIVATE tree_lab)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        if(TREE_LAB_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    elseif(TREE_LAB_SANITIZERS)
        message(FATAL_ERROR "Sanitizers require GCC or Clang")
    endif()
endfunction()

if(TREE_LAB_BUILD_EXAMPLES)
    add_executable(tree_lab_demo examples/main.cpp)
    tree_lab_configure_target(tree_lab_demo)
endif()

if(TREE_LAB_BUILD_BENCHMARKS)
    add_executable(tree_lab_benchmark benchmarks/benchmark.cpp)
    tree_lab_configure_target(tree_lab_benchmark)
endif()

if(TREE_LAB_BUILD_TESTS)
    enable_testing()
    find_package(GTest CONFIG QUIET)
    if(NOT GTest_FOUND)
        include(FetchContent)
        FetchContent_Declare(googletest
            URL https://github.com/google/googletest/archive/refs/tags/v1.17.0.tar.gz
            URL_HASH SHA256=65fab701d9829d38cb77c14acdc431d2108bfdbf8979e40eb8ae567edf10b27c
            DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
        set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
        set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
        FetchContent_MakeAvailable(googletest)
    endif()
    add_executable(tree_lab_tests tests/tree_test.cpp tests/set_test.cpp tests/containers_test.cpp)
    tree_lab_configure_target(tree_lab_tests)
    target_link_libraries(tree_lab_tests PRIVATE GTest::gtest_main)
    include(GoogleTest)
    gtest_discover_tests(tree_lab_tests)
endif()
